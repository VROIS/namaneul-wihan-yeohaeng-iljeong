    function formatRelativeTime(date) {
      const now = new Date();
      const diff = now - new Date(date);
      const minutes = Math.floor(diff / (1000 * 60));
      const hours = Math.floor(diff / (1000 * 60 * 60));
      const days = Math.floor(diff / (1000 * 60 * 60 * 24));
      
      if (minutes < 60) return `${minutes}분 전`;
      if (hours < 24) return `${hours}시간 전`;
      if (days < 7) return `${days}일 전`;
      if (days < 30) return `${Math.floor(days / 7)}주 전`;
      return `${Math.floor(days / 30)}개월 전`;
    }

    async function deleteApiKey(keyName, displayName) {
      if (!confirm(`"${displayName}"(${keyName})를 삭제하시겠습니까?\n이 키를 쓰는 기능이 즉시 멈출 수 있습니다.`)) return;

        const res = await fetch(`${API_BASE}/api-keys/${keyName}`, { method: 'DELETE' });
        if (res.ok) {
          loadApiKeys();

          const data = await res.json().catch(() => ({}));
          alert('삭제 실패: ' + (data.error || res.statusText));
        }
      } catch (err) {
        alert('삭제 중 오류: ' + err.message);
      }
    }

    async function loadContentOverrideCities() {
      const sel = document.getElementById('content-city-select');
      if (!sel || sel.dataset.loaded) return;
      try {
        const res = await fetch('/api/cities');
        const list = await res.json();
        list.sort((a, b) => (a.nameEn || a.name || '').localeCompare(b.nameEn || b.name || ''));
        sel.innerHTML = '<option value="">도시를 선택하세요</option>' +
          list.map(c => `<option value="${c.id}">${c.nameEn || c.name}</option>`).join('');
        sel.dataset.loaded = '1';
      } catch (err) {
        console.error('도시 목록 로드 오류:', err);
      }
    }

    async function loadContentPreview(cityId) {
      const nameEl = document.getElementById('content-preview-name');
      const taglineEl = document.getElementById('content-preview-tagline');
      const imgEl = document.getElementById('content-preview-image');
      const btn = document.getElementById('content-apply-override-btn');
      clearOverrideInputs();
      if (!cityId) {
        nameEl.textContent = '도시를 선택하세요';
        taglineEl.textContent = '-';
        btn.dataset.cityId = '';
        return;
      }
      try {
        const res = await fetch(`/api/cities/${cityId}/representative?lang=ko`);
        const card = await res.json();
        nameEl.textContent = `${card.nameKo || card.nameEn || ''} (${card.country || '-'})`;
        taglineEl.textContent = card.tagline || '(태그라인 없음)';
        if (card.imageUrl) {
          imgEl.style.backgroundImage = `url(${card.imageUrl})`;
          imgEl.textContent = '';
        } else {
          imgEl.style.backgroundImage = '';
          imgEl.textContent = '대표사진 없음';
        }
        document.getElementById('content-preview-badge-video').className =
          'status-badge ' + (card.hasVideo ? 'status-ok' : 'status-warn');
        document.getElementById('content-preview-badge-guide').className =
          'status-badge ' + (card.hasGuide ? 'status-ok' : 'status-warn');

        document.getElementById('content-preview-badge-route').className =
          'status-badge ' + (card.itineraryId !== null ? 'status-ok' : 'status-warn');

        // ⚠️ 2026-08-25 판단3종 지적으로 추가 = 지금 이 도시에 저장된 override 를 폼에 채워서 보여준다.
        //   안 채우면 슬롯 하나만 고치려 해도 나머지 저장값이 빈칸으로 보여, [적용] 시 그 5칸을 그대로 전송하는
        //   서버(전체 UPDATE)가 나머지 저장값을 조용히 지운다(§14 새것우선과 달리 이건 부분수정 파괴 버그였음).

        document.getElementById('override-hero-place-id').value = ov.heroPlaceId ?? '';
        const highlights = ov.highlightPlaceIds || [];
        document.getElementById('override-highlight-1').value = highlights[0] ?? '';
        document.getElementById('override-highlight-2').value = highlights[1] ?? '';
        document.getElementById('override-highlight-3').value = highlights[2] ?? '';

      } catch (err) {
        console.error('콘텐츠 미리보기 로드 오류:', err);
      }
    }

    async function applyContentOverride() {
      const btn = document.getElementById('content-apply-override-btn');
      const cityId = btn.dataset.cityId;
      if (!cityId) { alert('도시를 먼저 선택하세요'); return; }
      const val = (id) => {
        const v = document.getElementById(id).value.trim();
        return v ? Number(v) : null;
      };
      const body = {
        heroPlaceId: val('override-hero-place-id'),
        highlightPlaceIds: [
          val('override-highlight-1'), val('override-highlight-2'), val('override-highlight-3'),
        ].filter((v) => v != null),
        videoId: val('override-video-id'),
      };
      try {
        const res = await fetch(`${API_BASE}/cities/${cityId}/content-override`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(body),
        });
        if (!res.ok) throw new Error((await res.json()).error || res.statusText);
        await loadContentPreview(cityId);
        alert('적용 완료 — 도시카드에 즉시 반영됩니다.');
      } catch (err) {
        console.error('선별입력 저장 오류:', err);
        alert('저장 실패: ' + err.message);
      }
    }

    async function loadGuidePrices() {
      loadContentOverrideCities(); // 🖼️ 콘텐츠관리 탭 열 때 도시목록도 같이 채움(§16)

        prices.sort((a, b) => a.id - b.id);

              <button class="btn btn-primary" onclick="seedGuidePrices()" style="margin-top: 16px;">기본 가격 입력</button>
            </div>`;
          return;
        }
        
        const unitLabels = { day: '/일', trip: '/회', hour: '/시간' };

              <div style="font-weight: 600; font-size: 15px;">${p.serviceName}</div>

                ${p.features ? p.features.map(f => `<span style="background: rgba(255,255,255,0.05); padding: 2px 8px; border-radius: 4px; margin-right: 4px;">${f}</span>`).join('') : ''}
              </div>
            </div>
            <div style="text-align: right;">
              <div style="font-size: 20px; font-weight: 700; color: #a78bfa;">
                ${p.pricePerDay ? `€${p.pricePerDay}` : `€${p.priceLow}~${p.priceHigh}`}

    }

    function updatePriceFormula() {
      // 세단 8시간 계산
      const sedanBase = parseFloat(document.getElementById('sedan-base-price').value) || 240;
      const sedanHourly = parseFloat(document.getElementById('sedan-hourly-price').value) || 60;
      const sedan8h = sedanBase + (4 * sedanHourly); // 기본 4h + 추가 4h
      document.getElementById('sedan-8h-total').textContent = `€${sedan8h}`;
      document.getElementById('compare-sedan').textContent = `€${sedan8h}`;
      
      // 밴 8시간 계산
      const vanBase = parseFloat(document.getElementById('van-base-price').value) || 320;
      const vanHourly = parseFloat(document.getElementById('van-hourly-price').value) || 80;
      const van8h = vanBase + (4 * vanHourly);
      document.getElementById('van-8h-total').textContent = `€${van8h}`;
      document.getElementById('compare-van').textContent = `€${van8h}`;
      
      // 미니버스 8시간 계산
      const minibusBase = parseFloat(document.getElementById('minibus-base-price').value) || 400;
      const minibusHourly = parseFloat(document.getElementById('minibus-hourly-price').value) || 100;
      const minibus8h = minibusBase + (4 * minibusHourly);
      document.getElementById('minibus-8h-total').textContent = `€${minibus8h}`;
      
      // 가이드 온리 8시간 계산
      const guideHourly = parseFloat(document.getElementById('guide-only-hourly').value) || 50;
      const guide8h = guideHourly * 8;
      document.getElementById('guide-only-8h-total').textContent = `€${guide8h}`;
      
      updateComparisonNote();
    }
    
    function updateComparisonNote() {
      const sedan8h = parseFloat(document.getElementById('compare-sedan').textContent.replace('€', ''));
      const uberBlackSedan = document.getElementById('uber-black-sedan').value;
      
      // 우버 블랙 최고가 추출 (예: "€450~550" → 550)
      const uberHighMatch = uberBlackSedan.match(/(\d+)$/);
      const uberHigh = uberHighMatch ? parseInt(uberHighMatch[1]) : 550;
      
      const savings = uberHigh - sedan8h;
      if (savings > 0) {
        const noteEl = document.getElementById('comparison-marketing-note');
        if (noteEl && !noteEl.dataset.userEdited) {
          noteEl.value = `가이드 서비스가 우버 블랙 대비 최대 €${savings} 절약! 전문 가이드 동행으로 시간 효율 2배 ↑ 전일 대기로 호출 스트레스 없음`;
        }
      }
    }
    
    async function saveHourlyPrices() {
      const hourlyPrices = {
        sedan: {
          basePrice4h: parseFloat(document.getElementById('sedan-base-price').value) || 240,
          pricePerHour: parseFloat(document.getElementById('sedan-hourly-price').value) || 60,
          minPassengers: 1,
          maxPassengers: 4
        },
        van: {
          basePrice4h: parseFloat(document.getElementById('van-base-price').value) || 320,
          pricePerHour: parseFloat(document.getElementById('van-hourly-price').value) || 80,
          minPassengers: 5,
          maxPassengers: 7
        },
        minibus: {
          basePrice4h: parseFloat(document.getElementById('minibus-base-price').value) || 400,
          pricePerHour: parseFloat(document.getElementById('minibus-hourly-price').value) || 100,
          minPassengers: 8,
          maxPassengers: 20
        },
        guide_only: {
          basePrice4h: 0,
          pricePerHour: parseFloat(document.getElementById('guide-only-hourly').value) || 50,
          minPassengers: 1,
          maxPassengers: 20
        }
      };
      
      // 비교표 데이터
      const comparison = {
        uberBlack: {
          sedan: document.getElementById('uber-black-sedan').value,
          van: document.getElementById('uber-black-van').value
        },
        uberX: {
          sedan: document.getElementById('uber-x-sedan').value
        },
        taxi: {
          sedan: document.getElementById('taxi-sedan').value,
          van: document.getElementById('taxi-van').value
        },
        marketingNote: document.getElementById('comparison-marketing-note').value
      };
      
      try {
        const res = await fetch(`${API_BASE}/guide-prices/hourly`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ hourlyPrices, comparison })
        });
        
        if (res.ok) {
          const data = await res.json();
          alert('✅ 시간당 가격이 저장되었습니다!');
          loadGuidePrices();
        } else {
          const err = await res.json();
          alert('저장 실패: ' + (err.error || '알 수 없는 오류'));
        }
      } catch (err) {
        alert('저장 중 오류: ' + err.message);
      }
    }
    
    async function loadHourlyPrices() {
      try {
        const res = await fetch(`${API_BASE}/guide-prices/hourly`);
        if (!res.ok) return;
        
        const data = await res.json();
        
        if (data.sedan) {
          document.getElementById('sedan-base-price').value = data.sedan.basePrice4h || 240;
          document.getElementById('sedan-hourly-price').value = data.sedan.pricePerHour || 60;
        }
        if (data.van) {
          document.getElementById('van-base-price').value = data.van.basePrice4h || 320;
          document.getElementById('van-hourly-price').value = data.van.pricePerHour || 80;
        }
        if (data.minibus) {
          document.getElementById('minibus-base-price').value = data.minibus.basePrice4h || 400;
          document.getElementById('minibus-hourly-price').value = data.minibus.pricePerHour || 100;
        }
        if (data.guide_only) {
          document.getElementById('guide-only-hourly').value = data.guide_only.pricePerHour || 50;
        }
        
        // 비교표 로드
        if (data.comparison) {
          if (data.comparison.uberBlack) {
            document.getElementById('uber-black-sedan').value = data.comparison.uberBlack.sedan || '€450~550';
            document.getElementById('uber-black-van').value = data.comparison.uberBlack.van || '€600~750';
          }
          if (data.comparison.uberX) {
            document.getElementById('uber-x-sedan').value = data.comparison.uberX.sedan || '€280~350';
          }
          if (data.comparison.taxi) {
            document.getElementById('taxi-sedan').value = data.comparison.taxi.sedan || '€320~400';
            document.getElementById('taxi-van').value = data.comparison.taxi.van || '€450~550';
          }
          if (data.comparison.marketingNote) {
            document.getElementById('comparison-marketing-note').value = data.comparison.marketingNote;
            document.getElementById('comparison-marketing-note').dataset.userEdited = 'true';
          }
        }
        
        updatePriceFormula();
      } catch (err) {
        console.error('시간당 가격 로드 실패:', err);
      }
    }
    
    async function saveComparisonPrices() {
      await saveHourlyPrices(); // 같은 API 사용
    }
    
    // 가이드 가격 탭 로드시 시간당 가격도 로드
    const originalLoadGuidePrices = loadGuidePrices;
    loadGuidePrices = async function() {
      await originalLoadGuidePrices();
      await loadHourlyPrices();

    //   설명 §0-1 원칙). 시각 숫자만 남김.
    function updateRefreshTime() {
      const el = document.getElementById('last-update');
      if (el) {
        el.textContent = lastUpdateTime.toLocaleTimeString('ko-KR');
      }
    }
    
    async function refreshAll() {
      const activeTab = document.querySelector('.tab.active')?.dataset.tab || 'overview';
      await loadTabData(activeTab);
      lastUpdateTime = new Date();
      updateRefreshTime();
    }
    
    function startAutoRefresh() {
      if (autoRefreshInterval) clearInterval(autoRefreshInterval);
      autoRefreshInterval = setInterval(refreshAll, 30000);
      console.log('자동 새로고침 시작 (30초 간격)');
    }
    
    function stopAutoRefresh() {
      if (autoRefreshInterval) {
        clearInterval(autoRefreshInterval);
        autoRefreshInterval = null;
        console.log('자동 새로고침 중지');
      }
    }
    
    startAutoRefresh();
    
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) {
        stopAutoRefresh();
      } else {
        refreshAll();
        startAutoRefresh();
      }
    });
