} from "./route-types";

// ⚠️ 2026-07-04 사장님 SSOT = 드라이빙 가이드 판별 단일 SSOT(Gemini·MIX 경로와 동일 함수 = 3경로 정합, §16 재발명 금지).

  FREE_THRESHOLD_EUR,

} from "../shared/slot-duration";

type LatLng = { lat: number; lng: number };

const hasCoord = (p: { lat?: number | null; lng?: number | null }): boolean =>
  p.lat != null && p.lng != null && p.lat !== 0 && p.lng !== 0;

const toMin = (t: string): number => {
  const [h, m] = t.split(":").map(Number);
  return h * 60 + m;
};

const round2 = (n: number) => Math.round(n * 100) / 100;

export function buildRouteLocal(
  skeleton: AG1Output,
  places: PlaceResult[],
  cityCoords: LatLng | undefined,
  restaurantPool?: PlaceResult[],
  hourlyRate?: number | null,
  mealTiers?: CityMealTiers | null,
): RouteHandlerResult {
  const t0 = Date.now();
  const { formData, daySlotsConfig, paceConfig, companionCount } = skeleton;
  const slotDuration = paceConfig.slotDurationMinutes; // 활동 1곳 시간
  const mealDuration = paceConfig.mealDurationMinutes; // 식사 1회 시간(밀도별, 활동보다 짧음)
  // ⚠️ 수정금지(승인필요) 2026-08-31 사장님 확정 = 예산 구간 = 그 도시 분포 경계선(20/60/20) (정본 B4)
  //   = 전 도시 고정 유로(점심 상한 €16 등)는 물가를 못 담아 근처 후보를 전부 떨어뜨림 = 폐기 §19.
  //   = 표본 부족 도시만 MEAL_BUDGET 고정값으로 되돌린다.
  const style = normalizeTravelStyle(formData.travelStyle);
  const fixed = MEAL_BUDGET[style];
  const band = mealTiers
    ? tierRange(style, mealTiers)
    : { min: fixed.min, cap: fixed.max };
  // ⚠️ 수정금지(승인필요) 2026-07-31 사장님 승인(BTS D단계 BE-3) = 핀 식당 id 집합(풀 id 형식 = "db-<번호>").
  const pinnedRestIds = new Set(
    (formData.pinnedPlaceIds ?? []).map((n) => `db-${n}`),
  );

  // ⚠️ 2026-07-04 사장님 SSOT = 출발/종료 앵커 = 숙소 좌표 최우선(§14 A안=여행 전체 공통 숙소) → 없으면 도심 중심.
  const firstValid = places.find(hasCoord);
  const center: LatLng =
    formData.accommodationCoords && hasCoord(formData.accommodationCoords)
      ? formData.accommodationCoords
      : cityCoords && hasCoord(cityCoords)
        ? cityCoords
        : formData.destinationCoords && hasCoord(formData.destinationCoords)
          ? formData.destinationCoords
          : firstValid
            ? { lat: firstValid.lat, lng: firstValid.lng }
            : { lat: 0, lng: 0 };

  // ⚠️ 2026-07-04 사장님 SSOT = 드라이빙 가이드 = 이동(Minimal·Moderate) OR 예산(Premium·Luxury) 중 하나라도 = 무조건 가이드(본업 퍼널).
  const transport: "public_transit" | "private_driver_guide" =
    shouldApplyGuidePrice(formData.mobilityStyle, formData.travelStyle)
      ? "private_driver_guide"
      : "public_transit";

  // ⚠️ 수정금지(승인필요) 2026-09-01 사장님 확정 = 곳수 상한 없음 = 선정은 분(minute) 기준이 유일 (정본 B4)
  const candidates = places.filter(
    (p) => p.seedCategory !== "restaurant" && hasCoord(p),
  );
  const activities = [...candidates];

  const restaurants = (
    restaurantPool && restaurantPool.length
      ? restaurantPool
      : places.filter((p) => p.seedCategory === "restaurant")
  ).filter(hasCoord);

  const usedRest = new Set<string>();
  // ⚠️ 수정금지(승인필요) 2026-09-02 사장님 확정 = 식당 = 컷오프 없음·전체 대상·거리 최우선 = 가까운 반경부터, 그 안에서 예산 띠 → 상한 이하 → 베스트>RC. 점심 = 직전 활동, 저녁 = 실제 마지막 활동 (정본 B4 v26)
  const WALK_KM = 1;
  const pickMealPriority = (
    anchors: LatLng[],
    priceMin: number,
    priceCap: number,
  ): PlaceResult | null => {
    const refs = anchors.filter((a) => a && a.lat != null && a.lat !== 0);
    const useRefs = refs.length ? refs : [center];
    const scored = restaurants
      .filter((r) => !usedRest.has(r.id))
      .map((r) => ({
        r,
        d: Math.min(
          ...useRefs.map((a) => haversineKm(a.lat, a.lng, r.lat, r.lng)),
        ),
      }));
    if (!scored.length) return null;

    // ① 핀 식당 = 사용자가 직접 고른 것 = 모든 기준보다 우선(2026-07-31 사장님 승인).
    const pinnedPick = scored.find((x) => pinnedRestIds.has(x.r.id));
    if (pinnedPick) {
      usedRest.add(pinnedPick.r.id);
      return pinnedPick.r;
    }

    const lo = Math.min(priceMin, priceCap);
    const byBestRc = (a: (typeof scored)[0], b: (typeof scored)[0]) =>
      bestRankLangCount(b.r.bestRank) - bestRankLangCount(a.r.bestRank) ||
      (a.r.rank ?? 9999) - (b.r.rank ?? 9999) ||
      a.d - b.d;
    const inBudget = (x: (typeof scored)[0]) =>
      x.r.estimatedPriceEur != null &&
      x.r.estimatedPriceEur >= lo &&
      x.r.estimatedPriceEur <= priceCap;
    const underCap = (x: (typeof scored)[0]) =>
      x.r.estimatedPriceEur != null && x.r.estimatedPriceEur <= priceCap;

    const RINGS = [WALK_KM, 3, 8, Infinity];
    for (const radius of RINGS) {
      const ring = scored.filter((x) => x.d <= radius);
      if (!ring.length) continue;
      const inB = ring.filter(inBudget);
      const pool = inB.length ? inB : ring.filter(underCap);
      if (!pool.length) continue;
      const hit = pool.sort(byBestRc)[0];
      usedRest.add(hit.r.id);
      return hit.r;
    }
    const any = [...scored].sort((a, b) => a.d - b.d)[0];
    usedRest.add(any.r.id);
    return any.r;
  };

  const SAME_SPOT_KM = 0.5;
  // 같은 장소가 창고에 여러 행으로 있을 때(앵발리드 45m 실측)만 걸러낸다.
  // 거리만 보면 78m 떨어진 생트샤펠·콩시에르쥬리처럼 서로 다른 명소가 같이 잘린다.
  const keyOf = (p: PlaceResult) =>
    String(p.nameKo || p.name || "")
      .toLowerCase()
      .replace(/[^a-z0-9가-힣]/g, "");
  const sameName = (a: PlaceResult, b: PlaceResult) => {
    const x = keyOf(a),
      y = keyOf(b);
    return !!x && !!y && (x.includes(y) || y.includes(x));
  }; // 이 안 = 같은 자리(리조트·단지) = 같은 날에 묶음
  const OUTSKIRT_MAX_KM = 100;
  const kmFromCenter = (p: PlaceResult) =>
    haversineKm(center.lat, center.lng, p.lat, p.lng);
  // ⚠️ 수정금지(승인필요) 2026-09-01 사장님 확정 = 사용자가 고른 카테고리가 바깥 껍질 = 그 안에서 베스트→rank (정본 B4)
  //   = 껍질 없이 베스트부터 보면 힐링을 골랐는데 루브르(heritage ★7)가 1번으로 감(파리 5일 실측).
  const CAT_FALLBACK_ORDER = [
    "heritage",
    "attraction",
    "hotspot",
    "healing",
    "adventure",
    "shopping",
  ];
  const pickedCats = (formData.vibes ?? [])
    .map((v) => VIBE_PRIMARY_CATEGORY[v])
    .filter((c): c is string => !!c && c !== "restaurant");
  const catOrder = [
    ...pickedCats,
    ...CAT_FALLBACK_ORDER.filter((c) => !pickedCats.includes(c)),
  ];
  const catTier = (p: PlaceResult) => {
    const i = catOrder.indexOf(p.seedCategory ?? "");
    return i < 0 ? catOrder.length : i;
  };
  const byBestThenRank = (a: PlaceResult, b: PlaceResult) =>
    catTier(a) - catTier(b) ||
    bestRankLangCount(b.bestRank) - bestRankLangCount(a.bestRank) ||
    (a.rank ?? 9999) - (b.rank ?? 9999);

  // ⚠️ 수정금지(승인필요) 2026-08-31 사장님 확정 = ① 여행 전체에서 뽑을 곳을 먼저 확정(베스트 전부 → 카테고리별 RC) (정본 B4)
  //   = 날짜별로 그때그때 고르면 앞날이 자리를 다 써 뒤에 남은 만장일치가 통째 탈락함(파리 ★7 3곳 실측).
  // 자리 = 곳수가 아니라 분(minute) = 입장료 슬롯이 밀도값보다 길어 곳수로 세면 뒤쪽이 통째 탈락함.
  const stayMin = (p: PlaceResult) =>
    slotMinutesFor(
      p.estimatedPriceEur,
      slotDuration,
      hourlyRate ?? null,
      p.seedCategory,
    );
  // ⚠️ 수정금지(승인필요) 2026-09-01 사장님 확정 = 유료 입장지는 18:00 이후 시작 불가(대부분 폐관) (정본 B4)
  const PAID_LAST_START_MIN = 18 * 60;
  const isPaidPlace = (p: PlaceResult) =>
    (p.estimatedPriceEur ?? 0) > FREE_THRESHOLD_EUR &&
    PRICED_STAY_CATEGORIES.has(p.seedCategory ?? "");
  const dayActMinutes = daySlotsConfig.map(
    (dc) => toMin(dc.endTime) - toMin(dc.startTime) - mealDuration * 2,
  );
  const totalMinutes = dayActMinutes.reduce((s, v) => s + v, 0);
  const inRange = activities.filter((p) => kmFromCenter(p) <= OUTSKIRT_MAX_KM);
  // 2026-09-02 v26d = AI의걬(#478 무리) 반영 = 교외(도심 15km 밖)는 베스트만 후보(사장님 실증 판정 전)
  const farOk = (p: PlaceResult) =>
    kmFromCenter(p) <= DAYTRIP_KM || bestRankLangCount(p.bestRank) > 0;
  const isSameSpot = (a: PlaceResult, b: PlaceResult) =>
    haversineKm(a.lat, a.lng, b.lat, b.lng) <= SAME_SPOT_KM && sameName(a, b);
  // ⚠️ 수정금지(승인필요) 2026-09-02 사장님 확정 = 컷오프 = 사용자 바이브 바대로(AG1 vibeWeights) 카테고리 자리를 나누고, 각 카테고리 안에서 베스트 → rank. 남는 시간은 전체에서 베스트 → rank. 이동시간은 어디에도 더하지 않는다 (정본 B4 v26)
  const shareOf = new Map<string, number>();
  for (const vw of skeleton.vibeWeights ?? []) {
    const c = VIBE_PRIMARY_CATEGORY[vw.vibe];
    if (c && c !== "restaurant")
      shareOf.set(c, (shareOf.get(c) ?? 0) + vw.weight);
  }
  const shareSum = [...shareOf.values()].reduce((a, b) => a + b, 0) || 1;
  const quotaOf = new Map(
    [...shareOf].map(([c, w]) => [c, (totalMinutes * w) / shareSum] as const),
  );
  const usedOf = new Map<string, number>();
  const selected: PlaceResult[] = [];
  let usedMin = 0;
  const take = (p: PlaceResult) => {
    selected.push(p);
    usedMin += stayMin(p);
    const c = p.seedCategory ?? "";
    usedOf.set(c, (usedOf.get(c) ?? 0) + stayMin(p));
  };
  const canTake = (p: PlaceResult) =>
    !selected.includes(p) &&
    farOk(p) &&
    usedMin + stayMin(p) <= totalMinutes &&
    !selected.some((q) => isSameSpot(q, p));
  const ranked = [...inRange].sort(byBestThenRank);
  for (const p of ranked) {
    const c = p.seedCategory ?? "";
    const q = quotaOf.get(c);
    if (q == null || !canTake(p)) continue;
    if ((usedOf.get(c) ?? 0) + stayMin(p) > q) continue;
    take(p);
  }
  const byBestFirst = (a: PlaceResult, b: PlaceResult) =>
    bestRankLangCount(b.bestRank) - bestRankLangCount(a.bestRank) ||
    catTier(a) - catTier(b) ||
    (a.rank ?? 9999) - (b.rank ?? 9999);
  for (const p of [...inRange].sort(byBestFirst)) if (canTake(p)) take(p);

  // ⚠️ 수정금지(승인필요) 2026-09-02 사장님 확정 = 컷오프 통과분은 동등 = 구역·방향으로 날을 나누고(시간 균등), 매일 가장 먼 곳에서 출발해 도심으로 귀소 (정본 B4 v26)
  const dayLists = buildZoneDays(selected, center, dayActMinutes, stayMin);
  // ⚠️ 수정금지(승인필요) 2026-09-01 사장님 확정 = 점심 12:00 뒤 첫 자리 · 저녁 19:00 이후 · 5시간 이상 머무는 곳은 점심 생략 (정본 B4)
  const LUNCH_FROM = 12 * 60;
  const LONG_STAY_MIN = 300;
  const DINNER_FROM = 19 * 60;
  const hasRest = restaurants.length > 0;
  type DayCfg = (typeof daySlotsConfig)[number];
  const schedule = (order: PlaceResult[], dc: DayCfg) => {
    const endMin = toMin(dc.endTime);
    const deadline = hasRest ? endMin - mealDuration : endMin;
    let acc = toMin(dc.startTime);
    let lunchDone = !hasRest;
    const placed: PlaceResult[] = [];
    const late: PlaceResult[] = [];
    for (const a of order) {
      if (!lunchDone && acc >= LUNCH_FROM) {
        acc = Math.max(acc, LUNCH_FROM) + mealDuration;
        lunchDone = true;
      }
      const d = stayMin(a);
      if (
        (isPaidPlace(a) && acc >= PAID_LAST_START_MIN) ||
        acc + d > deadline
      ) {
        late.push(a);
        continue;
      }
      placed.push(a);
      acc += d;
      if (!lunchDone && d >= LONG_STAY_MIN) lunchDone = true;
    }
    return { placed, late };
  };
  // 2026-09-02 v26 시뮬 보정(사장님 실증 판정 전) = 18시 뒤로 밀린 유료는 그 3km 안 무료 하나를 뒤로 미뤄 앞당긴다
  const repairPaid = (order: PlaceResult[], dc: DayCfg) => {
    const cur = [...order];
    for (let guard = 0; guard < cur.length * cur.length; guard++) {
      const p = schedule(cur, dc).late.find(isPaidPlace);
      if (!p) break;
      const idx = cur.indexOf(p);
      let f = -1;
      for (let i = idx - 1; i >= 0; i--)
        if (
          !isPaidPlace(cur[i]) &&
          haversineKm(cur[i].lat, cur[i].lng, p.lat, p.lng) <= 3
        ) {
          f = i;
          break;
        }
      if (f < 0) break;
      const [free] = cur.splice(f, 1);
      cur.splice(cur.indexOf(p) + 1, 0, free);
    }
    return cur;
  };
  const placedActs = new Set<string>();
  const dayOrders: PlaceResult[][] = [];
  let carry: PlaceResult[] = [];
  let dropped: PlaceResult[] = [];
  for (let di = 0; di < daySlotsConfig.length; di++) {
    const order = repairPaid(
      homePath([...carry, ...(dayLists[di] ?? [])], center),
      daySlotsConfig[di],
    );
    const { placed, late } = schedule(order, daySlotsConfig[di]);
    dayOrders.push(placed);
    placed.forEach((p) => placedActs.add(p.id));
    carry = di < daySlotsConfig.length - 1 ? late : [];
    if (di === daySlotsConfig.length - 1) dropped = late;
  }
  // 2026-09-02 v26 시뮬 보정(사장님 실증 판정 전) = 끼워 넣기 = 그 날 동선에 가장 싸게(km) 넣되 시각이 안 깨질 때만. ① 넘친 곳 ② 남는 자리는 창고에서 가까운 곳(고른 카테고리 먼저) = 빈 슬롯 0
  const tryInsert = (di: number, a: PlaceResult): boolean => {
    const cur = dayOrders[di];
    if (tripCount([...cur, a], center) > 1) return false;
    const o = repairPaid(homePath([...cur, a], center), daySlotsConfig[di]);
    if (schedule(o, daySlotsConfig[di]).late.length) return false;
    dayOrders[di] = o;
    placedActs.add(a.id);
    return true;
  };
  for (const a of dropped)
    for (let di = 0; di < dayOrders.length; di++) if (tryInsert(di, a)) break;
  const isDup = (a: PlaceResult) =>
    placedActs.has(a.id) ||
    dayOrders.some((o) => o.some((q) => isSameSpot(q, a)));
  const insertKm = (o: PlaceResult[], a: PlaceResult) => {
    let m = Infinity;
    for (let pos = 0; pos <= o.length; pos++) {
      const prev = pos ? o[pos - 1] : center;
      const next = pos < o.length ? o[pos] : center;
      m = Math.min(
        m,
        haversineKm(prev.lat, prev.lng, a.lat, a.lng) +
          haversineKm(a.lat, a.lng, next.lat, next.lng) -
          haversineKm(prev.lat, prev.lng, next.lat, next.lng),
      );
    }
    return m;
  };
  const tierBucket = (p: PlaceResult) =>
    catTier(p) < pickedCats.length ? 0 : 1;
  for (let di = 0; di < dayOrders.length; di++) {
    for (;;) {
      const pool = inRange
        .filter((a) => !isDup(a) && farOk(a))
        .map((a) => ({ a, k: insertKm(dayOrders[di], a) }))
        .sort((x, y) => tierBucket(x.a) - tierBucket(y.a) || x.k - y.k)
        .slice(0, 30);
      if (!pool.some(({ a }) => tryInsert(di, a))) break;
    }
  }

  const days: RouteResponse["days"] = [];
  let grandKm = 0;
  let grandSec = 0;

  for (let di = 0; di < daySlotsConfig.length; di++) {
    const dc = daySlotsConfig[di];
    const endMin = toMin(dc.endTime);
    const actDeadline = hasRest ? endMin - mealDuration : endMin;
    const seq: { p: PlaceResult; rest: boolean }[] = [];
    const slotStartMins: number[] = [];
    const slotDurations: number[] = [];
    let acc = toMin(dc.startTime);
    let cursor: LatLng | null = null;
    let lunchDone = !hasRest;
    // ⚠️ 수정금지(승인필요) 2026-09-02 사장님 확정 = 식사는 다음 활동의 기준점을 옮기지 않는다 (정본 B4 v26)
    const place = (p: PlaceResult, rest: boolean, dur: number, at: number) => {
      seq.push({ p, rest });
      slotStartMins.push(at);
      slotDurations.push(dur);
      acc = at + dur;
      if (!rest) cursor = { lat: p.lat, lng: p.lng };
    };
    const takeLunch = () => {
      lunchDone = true;
      const lunch = pickMealPriority([cursor ?? center], band.min, band.cap);
      if (!lunch) return;
      const at = Math.max(acc, LUNCH_FROM);
      if (at + mealDuration > actDeadline) return;
      place(lunch, true, mealDuration, at);
    };
    for (const a of dayOrders[di]) {
      if (!lunchDone && acc >= LUNCH_FROM) takeLunch();
      const dur = stayMin(a);
      place(a, false, dur, acc);
      if (!lunchDone && dur >= LONG_STAY_MIN) lunchDone = true;
    }
    if (!lunchDone) takeLunch();
    if (hasRest) {
      const dinner = pickMealPriority([cursor ?? center], band.min, band.cap);
      if (dinner) {
        const at = Math.max(acc, DINNER_FROM);
        if (at + mealDuration <= endMin) place(dinner, true, mealDuration, at);
      }
    }
    let prev: LatLng = center;
    let dayKm = 0;
    const scenes: RouteScene[] = seq.map((it, i) => {
      const km = round2(haversineKm(prev.lat, prev.lng, it.p.lat, it.p.lng));
      const { mode, calc } = pickTransitMode(
        km,
        transport === "private_driver_guide",
      );
      // ⚠️ 2026-07-04 사장님 SSOT = center(도시중심)를 넘겨 DRIVE 모드 도심/외곽 속도 분기(30/70km/h) 적용.
      const tr = calcTransitHaversine(
        prev,
        { lat: it.p.lat, lng: it.p.lng },
        calc,
        companionCount,
        center,
      );
      prev = { lat: it.p.lat, lng: it.p.lng };
      dayKm += km;
      grandSec += slotDurations[i] * 60;
      const scene: RouteScene = {
        slot: i + 1,
        time: minutesToTime(slotStartMins[i]),
        type: it.rest ? "restaurant" : "activity",
        place_id: it.p.id,
        name_en: it.p.name || "",
        name_ko: it.p.nameKo || "",
        name_local: it.p.nameLocal || it.p.name || "",
        address: it.p.address || "",
        lat: it.p.lat,
        lng: it.p.lng,
        distance_from_prev_km: km,
        transit_mode: mode,
        transit_min: tr.duration,
        slot_min: slotDurations[i],
      };
      if (it.rest && it.p.estimatedPriceEur != null)
        scene.price_eur = it.p.estimatedPriceEur;
      if (it.p.summaryKo) scene.selection_reason_ko = it.p.summaryKo;
      if (it.p.editorialSummary) scene.shortform_ko = it.p.editorialSummary;
      if (it.p.image) scene.image = it.p.image; // ⚠️ 2026-06-12 = PSR image_url 전달 (= 식당풀 픽 이미지 단절 해소)
      return scene;
    });

    days.push({ day: dc.day, total_distance_km: round2(dayKm), scenes });
    grandKm += dayKm;
  }

  const response: RouteResponse = {
    total_duration_sec: grandSec,
    total_distance_km: round2(grandKm),
    days,
  };
  return {
    ok: days.length > 0,
    response,
    raw: "",
    finishReason: "local-nn-haversine-v2",
    elapsedMs: Date.now() - t0,
  };
}
