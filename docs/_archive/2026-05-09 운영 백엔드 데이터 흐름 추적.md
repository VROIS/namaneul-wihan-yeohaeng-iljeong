```ts
async function generateNubiReasonV2(placeId, placeName, cityName, celebrityVisit, mergedData, seedData) {
  // P0: place_seed_raw DB 캐시 (1초 즉시 반환)
  if (seedData && seedData.nubiReason) return seedData.nubiReason;
  // 1순위: 셀럽 방문
  if (celebrityVisit && celebrityVisit.found) return `${celebrity} ${date} 게시`;
  // 2순위: 셀럽 멘션
  if (seedData && seedData.celebMention) return seedData.celebMention;
  // 3순위: 네이버 블로그
  if (seedData && seedData.naverBlogCount > 0) return `네이버 블로그 ${count}건`;
  // 4순위: 패키지투어
  if (mergedData.isPackageTourIncluded) {
    return `${packageMentionedBy.slice(0,2).join('·')} 필수코스`;
  }
  // 6순위: 구글 리뷰 ≥ 50
  const reviewCount = mergedData.userRatingCount || 0;
  if (reviewCount >= 50) return `구글 리뷰 ${reviewCount.toLocaleString()}개`;
  // Fallback
  return '데이터 수집 중';
}
```
