PhotoMedia 호출 (= post-process.ts 안 = 단일 관문 tsPhoto()):

import { tsPhoto } from 'server/services/shared/ts-client';

const imageUrl = await tsPhoto({
  apiKey: ${GOOGLE_PLACES_API_KEY},
  photoName,                        // = TsPlace.photoName (= "places/{id}/photos/{photo_id}")
  pathKey: `${cityId}/${category}/${pid}`,  // = .jpg 자동 부착 (PID 결정적 경로 = 2026-06-14 SSOT)
  // maxWidthPx 미지정 = 관문 기본 PHOTO_MAX_WIDTH_PX(400) = 해상도 단일 SSOT(2026-07-09)
});
